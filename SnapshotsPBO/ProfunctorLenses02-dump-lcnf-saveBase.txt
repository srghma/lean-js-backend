════ 📦 preview_left
def preview_left α β x.1 : Option α :=
  cases x.1 : Option α
  | Except.error a.2 =>
    let _x.3 := @Option.some _ a.2;
    return _x.3
  | Except.ok a.4 =>
    let _x.5 := @Option.none _;
    return _x.5

def preview_left_right α β γ x.1 : Option β :=

    cases a.2 : Option β

      return _x.4
    | _ =>
      let _x.5 := @Option.none _;
      return _x.5
  | _ =>
