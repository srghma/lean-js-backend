-- optimiser ran; the `.js` file beside this one holds what the optimiser made of

-- of the binder it belongs to, and every binding form names the types it binds.

        (var 1 : (list nat))
      )
      (jsOp cast nat → (fn [typeParam] nat)
        (var 1 : nat)
      )
    )
  )
)
